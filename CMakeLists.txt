cmake_minimum_required(VERSION 3.21)
project(RAMDPDF VERSION 0.9.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Pdf PdfWidgets PrintSupport Concurrent)
find_package(PkgConfig REQUIRED)
pkg_check_modules(POPPLER REQUIRED IMPORTED_TARGET poppler-qt6)
qt_standard_project_setup()
set(CMAKE_AUTORCC ON)
qt_add_executable(ramd-pdf src/main.cpp src/editor.cpp src/pdf_tools.cpp assets/resources.qrc)
target_link_libraries(ramd-pdf PRIVATE Qt6::Widgets Qt6::Pdf Qt6::PdfWidgets Qt6::PrintSupport Qt6::Concurrent PkgConfig::POPPLER)
install(TARGETS ramd-pdf RUNTIME DESTINATION bin)
install(FILES packaging/ramd-pdf.desktop DESTINATION share/applications)

install(FILES assets/ramd-pdf.png DESTINATION share/icons/hicolor/256x256/apps)
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(editor-tests tests/editor_tests.cpp src/editor.cpp)
  target_include_directories(editor-tests PRIVATE src)
  target_link_libraries(editor-tests PRIVATE Qt6::Widgets PkgConfig::POPPLER)
  add_test(NAME pdf-save-roundtrip COMMAND editor-tests)
  set_tests_properties(pdf-save-roundtrip PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  qt_add_executable(pdf-tools-tests tests/pdf_tools_tests.cpp src/pdf_tools.cpp src/editor.cpp)
  target_include_directories(pdf-tools-tests PRIVATE src)
  target_link_libraries(pdf-tools-tests PRIVATE Qt6::Widgets PkgConfig::POPPLER)
  add_test(NAME pdf-tools COMMAND pdf-tools-tests)
  qt_add_executable(reader-ui-tests tests/reader_ui_tests.cpp src/pdf_tools.cpp src/editor.cpp assets/resources.qrc)
  target_link_libraries(reader-ui-tests PRIVATE Qt6::Widgets Qt6::Pdf Qt6::PdfWidgets Qt6::PrintSupport Qt6::Concurrent PkgConfig::POPPLER)
  add_test(NAME reader-workspace COMMAND reader-ui-tests ${CMAKE_BINARY_DIR}/ui-previews)
  set_tests_properties(pdf-tools reader-workspace PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
if(WIN32)
  set_target_properties(ramd-pdf PROPERTIES WIN32_EXECUTABLE TRUE)
  file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/ramd-icon.rc "IDI_ICON1 ICON \"${CMAKE_CURRENT_SOURCE_DIR}/assets/ramd-pdf.ico\"\n")
  target_sources(ramd-pdf PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/ramd-icon.rc)
  set(CPACK_GENERATOR ZIP)
else()
  set(CPACK_GENERATOR DEB)
endif()
set(CPACK_PACKAGE_NAME "ramd-pdf")
set(CPACK_RPM_PACKAGE_NAME "ramd-pdf")
set(CPACK_RPM_PACKAGE_REQUIRES "qpdf, qt6-qtbase-gui, qt6-qtwayland")
set(CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION "/usr/share/icons" "/usr/share/icons/hicolor" "/usr/share/icons/hicolor/256x256" "/usr/share/icons/hicolor/256x256/apps")
set(CPACK_PACKAGE_CONTACT "RAMD Studios")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Offline PDF reader and annotation editor")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "RAMD Studios")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "qpdf")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
include(CPack)
