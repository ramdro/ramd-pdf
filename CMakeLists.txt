cmake_minimum_required(VERSION 3.21)
project(RAMDPDF VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Pdf PdfWidgets PrintSupport)
find_package(PkgConfig REQUIRED)
pkg_check_modules(POPPLER REQUIRED IMPORTED_TARGET poppler-qt6)
qt_standard_project_setup()
set(CMAKE_AUTORCC ON)
qt_add_executable(ramd-pdf src/main.cpp src/editor.cpp assets/resources.qrc)
target_link_libraries(ramd-pdf PRIVATE Qt6::Widgets Qt6::Pdf Qt6::PdfWidgets Qt6::PrintSupport PkgConfig::POPPLER)
install(TARGETS ramd-pdf RUNTIME DESTINATION bin)
if(NOT WIN32)
  install(FILES packaging/ramd-pdf.desktop DESTINATION share/applications)
endif()

install(FILES assets/ramd-pdf.png DESTINATION share/icons/hicolor/256x256/apps)
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(editor-tests tests/editor_tests.cpp src/editor.cpp)
  target_include_directories(editor-tests PRIVATE src)
  target_link_libraries(editor-tests PRIVATE Qt6::Widgets PkgConfig::POPPLER)
  add_test(NAME pdf-save-roundtrip COMMAND editor-tests)
  set_tests_properties(pdf-save-roundtrip PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
if(WIN32)
  set_target_properties(ramd-pdf PROPERTIES WIN32_EXECUTABLE TRUE)
  set(CPACK_GENERATOR ZIP)
else()
  set(CPACK_GENERATOR DEB)
endif()
set(CPACK_PACKAGE_CONTACT "RAMD Studios")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Offline PDF reader and annotation editor")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "RAMD Studios")
set(CPACK_DEBIAN_PACKAGE_DEPENDS "qpdf")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
include(CPack)
